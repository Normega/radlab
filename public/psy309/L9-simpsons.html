# =====================================================================
# PSY309 · Lecture 8 · Build Simpson's Paradox
#
# You will build a dataset in which studying MORE is associated with
# WORSE grades -- while, inside every single department, studying more
# is associated with BETTER grades.
#
# Both statements will be true of the same data.
# =====================================================================

set.seed(309)
n_per_dept <- 40

# ---- 1. Three departments -------------------------------------------
# The departments differ in two ways at once, and that is the whole
# trick:
#   - students in some departments study more hours on average
#   - those same departments grade harder
#
# Within each department, studying more helps. Always. We are about to
# write that in as a positive slope.

make_dept <- function(mean_hours, mean_grade, id) {
  hours  <- rnorm(n_per_dept, mean_hours, 1.6)
  grades <- mean_grade + 1.8 * (hours - mean_hours) + rnorm(n_per_dept, 0, 3)
  #                      ^^^ POSITIVE: more hours -> better grades
  data.frame(hours, grades, dept = factor(id))
}

d <- rbind(
  make_dept(12, 62, "A"),   # heavy workload, harsh grading
  make_dept( 8, 71, "B"),
  make_dept( 4, 83, "C")    # light workload, generous grading
)

# ---- 2. Look at the pooled data (the mistake) -----------------------
plot(d$hours, d$grades, pch = 19, col = rgb(0, 0, 0, .4),
     main = "All students pooled", xlab = "hours studied", ylab = "grade")
abline(lm(grades ~ hours, data = d), lwd = 3)

cor(d$hours, d$grades)
